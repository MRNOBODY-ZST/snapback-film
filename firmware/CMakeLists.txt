cmake_minimum_required(VERSION 3.16)
project(rb67_host_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(rb67_core src/controller.cpp src/journal.cpp)
target_include_directories(rb67_core PUBLIC src)
if(MSVC)
  target_compile_options(rb67_core PRIVATE /W4 /WX)
else()
  target_compile_options(rb67_core PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
foreach(check journal_tests controller_tests)
  add_executable(${check} tests/${check}.cpp)
  target_link_libraries(${check} PRIVATE rb67_core)
  add_test(NAME ${check} COMMAND ${check})
endforeach()
add_executable(host_sim host/sim.cpp)
target_link_libraries(host_sim PRIVATE rb67_core)
add_test(NAME host_sim COMMAND host_sim ${CMAKE_CURRENT_BINARY_DIR}/synthetic-transport.csv)
